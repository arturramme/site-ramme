import { Coletor } from '@ramme/vectorhub';
import { enfileirar, drenar } from './fila';

export type Apontamento = {
  ordem: string;
  posto: number;
  quantidade: number;
  turno: 'A' | 'B' | 'C';
};

type Resultado =
  | { ok: true; id: string }
  | { ok: false; motivo: string };

const conectados = new Map<string, Coletor>();
const RETENTATIVAS = 3;

export async function registrar(
  a: Apontamento,
): Promise<Resultado> {
  const agente = await Coletor.conectar();

  if (!agente.online) {
    // chao de fabrica nao para por causa da rede
    await enfileirar(a);
    return { ok: false, motivo: 'offline' };
  }

  for (let i = 0; i < RETENTATIVAS; i++) {
    const r = await agente.enviar({
      ...a,
      em: new Date().toISOString(),
    });

    if (r.aceito) return { ok: true, id: r.id };
  }

  return { ok: false, motivo: 'sem confirmacao' };
}

export async function sincronizar() {
  const pendentes = await drenar();
  return Promise.all(pendentes.map(registrar));
}
// consolida por ordem antes de subir para a nuvem
export function agrupar(lista: Apontamento[]) {
  return lista.reduce<Record<string, number>>((acc, a) => {
    acc[a.ordem] = (acc[a.ordem] ?? 0) + a.quantidade;
    return acc;
  }, {});
}

export const rotulo = (a: Apontamento) =>
  `${a.ordem}/${a.posto}`.padStart(12, '0');
